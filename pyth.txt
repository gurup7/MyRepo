"""
Secrets Scanner — Detects hardcoded credentials and sensitive variable names.

Based on Citi Checkmarx "Plaintext Storage of Sensitive Information" detection rules.
Scans: *.java, *.properties, *.yml, *.yaml, *.xml, *.sh, Dockerfile, docker-compose*

Usage:
    python secrets_scanner.py C:\\path\\to\\repo
    python secrets_scanner.py                      (scans current directory)

Output:
    - Console summary with all findings
    - CSV report auto-generated: secrets_scan_report_<repo_name>.csv
"""

import argparse
import csv
import os
import re
import sys
from dataclasses import dataclass
from pathlib import Path
from typing import List

# ============================================================================
# CONFIGURATION — Based on secrets-detection SKILL.md
# ============================================================================

# File patterns to scan
SCAN_PATTERNS = [
    "*.java",
    "*.properties",
    "*.yml",
    "*.yaml",
    "*.xml",
    "*.sh",
    "Dockerfile",
    "docker-compose*.yml",
    "docker-compose*.yaml",
]

# Directories to skip
EXCLUDE_DIRS = {
    ".git",
    "target",
    "build",
    "node_modules",
    ".mvn",
    ".gradle",
    ".idea",
    "bin",
    "obj",
    "__pycache__",
}

# Sensitive keywords in variable names (Checkmarx triggers)
SENSITIVE_KEYWORDS = [
    "SECRET",
    "ACCESSKEY",
    "ACCESS_KEY",
    "PASSWORD",
    "PASSWD",
    "TOKEN",
    "CREDENTIAL",
    "PRIVATE_KEY",
]

# ============================================================================
# DETECTION RULES
# ============================================================================

@dataclass
class Finding:
    file: str
    line: int
    finding_type: str  # HARDCODED_SECRET, SENSITIVE_VARNAME, EXPOSED_IN_LOG, DIRECT_INJECTION
    severity: str      # CRITICAL, HIGH, MEDIUM, LOW
    matched_text: str
    description: str
    remediation: str


def build_sensitive_varname_pattern():
    """Regex to detect variable names containing sensitive keywords."""
    keywords = "|".join(SENSITIVE_KEYWORDS)
    return re.compile(
        rf'\b\w*({keywords})\w*\b',
        re.IGNORECASE
    )


# Pre-compiled regex patterns
PATTERNS = {
    # Hardcoded UUID (access key format)
    "uuid_secret": {
        "regex": re.compile(
            r'[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}',
            re.IGNORECASE
        ),
        "type": "HARDCODED_SECRET",
        "severity": "CRITICAL",
        "description": "UUID-format value — potential access key or secret",
        "remediation": "Replace with NGC (CyberArk) secret nickname",
    },
    # AWS access key pattern
    "aws_key": {
        "regex": re.compile(r'(AKIA|ASIA)[A-Z0-9]{16}'),
        "type": "HARDCODED_SECRET",
        "severity": "CRITICAL",
        "description": "AWS access key pattern detected",
        "remediation": "Remove immediately. Use IAM roles or vault integration",
    },
    # Private key markers
    "private_key": {
        "regex": re.compile(r'-----BEGIN\s+(RSA\s+|EC\s+|DSA\s+)?PRIVATE KEY-----'),
        "type": "HARDCODED_SECRET",
        "severity": "CRITICAL",
        "description": "Private key embedded in source file",
        "remediation": "Move to secure vault. Never store private keys in source control",
    },
    # Password in properties/yaml (key=value or key: value)
    "password_value": {
        "regex": re.compile(
            r'(password|passwd|pwd)\s*[:=]\s*[^\s${}\[\]#\"\']{4,}',
            re.IGNORECASE
        ),
        "type": "HARDCODED_SECRET",
        "severity": "CRITICAL",
        "description": "Hardcoded password value in configuration",
        "remediation": "Replace with ${ENV_VAR} placeholder or NGC nickname",
    },
    # Connection string with embedded credentials
    "connection_string": {
        "regex": re.compile(
            r'(jdbc|mongodb|mongodb\+srv|amqp|redis|mysql|postgresql):\/\/[^:]+:[^@\s]+@',
            re.IGNORECASE
        ),
        "type": "HARDCODED_SECRET",
        "severity": "CRITICAL",
        "description": "Connection string with embedded credentials",
        "remediation": "Separate credential from URL. Use spring.datasource.password with vault",
    },
    # Base64 encoded strings (40+ chars, likely secrets)
    "base64_secret": {
        "regex": re.compile(r'[A-Za-z0-9+/]{40,}={0,2}'),
        "type": "HARDCODED_SECRET",
        "severity": "HIGH",
        "description": "Long Base64-encoded string — potential encoded secret",
        "remediation": "Verify if this is a credential. If yes, move to vault",
    },
    # Sensitive variable names (Checkmarx keyword triggers)
    # NOTE: In Java files, method names like getAccessToken() are typically
    # false positives. The real concern is config/Helm/K8s/shell files where
    # variable names appear in committed infra-as-code.
    "sensitive_varname": {
        "regex": build_sensitive_varname_pattern(),
        "type": "SENSITIVE_VARNAME",
        "severity": "HIGH",
        "description": "Variable name contains sensitive keyword (Checkmarx trigger)",
        "remediation": "Rename variable to remove sensitive keyword (see rename table)",
    },
    # Secrets echoed/printed to stdout (shell scripts)
    "secret_in_log": {
        "regex": re.compile(
            r'(echo|print|printf|log|logger)\s+.*\$\{?\w*(SECRET|PASSWORD|TOKEN|KEY)\w*\}?',
            re.IGNORECASE
        ),
        "type": "EXPOSED_IN_LOG",
        "severity": "MEDIUM",
        "description": "Sensitive variable potentially printed to stdout/logs",
        "remediation": "Log the variable NAME only, never the resolved VALUE",
    },
    # Direct env var injection in K8s/Helm
    "direct_injection": {
        "regex": re.compile(
            r'value:\s*\{\{\s*\.Values\.\w*(SECRET|ACCESSKEY|PASSWORD|TOKEN)\w*\s*\}\}',
            re.IGNORECASE
        ),
        "type": "DIRECT_INJECTION",
        "severity": "HIGH",
        "description": "Secret injected directly as env var in deployment manifest",
        "remediation": "Pass NGC nickname only. Resolve actual secret at runtime via O2C sidecar",
    },
}

# ============================================================================
# FALSE POSITIVE FILTERS
# ============================================================================

# Lines that are comments or known safe patterns
COMMENT_PATTERNS = [
    re.compile(r'^\s*#'),         # Shell/YAML comments
    re.compile(r'^\s*//'),        # Java/JS comments
    re.compile(r'^\s*\*'),        # Multi-line comment body
    re.compile(r'^\s*/\*'),       # Multi-line comment start
    re.compile(r'^\s*<!--'),      # XML comment
]

# Known false positive patterns (NGC nicknames references, etc.)
FALSE_POSITIVE_PATTERNS = [
    re.compile(r'NGC.*nickname', re.IGNORECASE),
    re.compile(r'CUR_\d+_\w+_\w+'),              # NGC nickname format
    re.compile(r'ngc\s+getSecret', re.IGNORECASE),  # NGC retrieval command
    re.compile(r'\$\{[^}]*:-}'),                    # Empty default placeholder ${VAR:-}
    re.compile(r'secretNickname'),                   # NGC parameter name
]

# File paths that are test/documentation (lower severity)
TEST_PATH_PATTERNS = [
    re.compile(r'test', re.IGNORECASE),
    re.compile(r'spec', re.IGNORECASE),
    re.compile(r'mock', re.IGNORECASE),
    re.compile(r'__tests__'),
    re.compile(r'fixtures'),
]


def is_comment(line: str) -> bool:
    """Check if a line is a comment."""
    return any(p.match(line) for p in COMMENT_PATTERNS)


def is_false_positive(line: str) -> bool:
    """Check if a line matches known false positive patterns."""
    return any(p.search(line) for p in FALSE_POSITIVE_PATTERNS)


def is_test_file(filepath: str) -> bool:
    """Check if file is in a test directory."""
    return any(p.search(filepath) for p in TEST_PATH_PATTERNS)


def is_java_file(filepath: str) -> bool:
    """Check if file is a Java source file."""
    return filepath.endswith(".java")


# ============================================================================
# SCANNER
# ============================================================================

def should_scan_file(filepath: Path) -> bool:
    """Check if file matches scan patterns and is not excluded."""
    # Check excluded directories
    parts = filepath.parts
    if any(excluded in parts for excluded in EXCLUDE_DIRS):
        return False

    # Check file pattern match
    name = filepath.name
    for pattern in SCAN_PATTERNS:
        if pattern.startswith("*"):
            if name.endswith(pattern[1:]):
                return True
        elif "*" in pattern:
            # Handle patterns like docker-compose*.yml
            prefix, suffix = pattern.split("*", 1)
            if name.startswith(prefix) and name.endswith(suffix):
                return True
        else:
            if name == pattern:
                return True
    return False


def scan_file(filepath: Path, repo_root: Path) -> List[Finding]:
    """Scan a single file for secrets."""
    findings = []
    relative_path = str(filepath.relative_to(repo_root))

    try:
        with open(filepath, "r", encoding="utf-8", errors="ignore") as f:
            lines = f.readlines()
    except (IOError, OSError):
        return findings

    for line_num, line in enumerate(lines, start=1):
        # Skip comments
        if is_comment(line):
            continue

        # Skip known false positives
        if is_false_positive(line):
            continue

        # Run each detection pattern
        for rule_name, rule in PATTERNS.items():
            matches = rule["regex"].finditer(line)
            for match in matches:
                matched_text = match.group(0).strip()

                # Skip if matched text is too short (likely false positive)
                if len(matched_text) < 4:
                    continue

                # Skip base64 pattern in binary-looking content or import statements
                if rule_name == "base64_secret":
                    if "import " in line or "package " in line or "class " in line:
                        continue
                    # Skip if it looks like a Java class path or XML namespace
                    if "." in matched_text and "/" not in matched_text:
                        continue

                # Adjust severity for test files
                severity = rule["severity"]
                if is_test_file(relative_path):
                    severity = "LOW"

                # In Java files, sensitive VARNAME findings are typically false positives
                # (method names like getAccessToken, getSecret are standard patterns)
                # Downgrade to LOW unless the line contains a hardcoded value assignment
                if rule_name == "sensitive_varname" and is_java_file(relative_path):
                    # Only flag as HIGH if it looks like a hardcoded assignment
                    # e.g., String SECRET = "actual-value"
                    if not re.search(r'=\s*"[^"]{8,}"', line):
                        severity = "LOW"

                finding = Finding(
                    file=relative_path,
                    line=line_num,
                    finding_type=rule["type"],
                    severity=severity,
                    matched_text=matched_text[:80],  # Truncate for readability
                    description=rule["description"],
                    remediation=rule["remediation"],
                )
                findings.append(finding)

    return findings


def scan_repository(repo_path: str) -> List[Finding]:
    """Scan entire repository for secrets."""
    repo_root = Path(repo_path).resolve()
    if not repo_root.is_dir():
        print(f"ERROR: {repo_path} is not a valid directory")
        sys.exit(1)

    all_findings = []
    files_scanned = 0

    for filepath in repo_root.rglob("*"):
        if filepath.is_file() and should_scan_file(filepath):
            files_scanned += 1
            findings = scan_file(filepath, repo_root)
            all_findings.extend(findings)

    print(f"\nScanned {files_scanned} files in {repo_root}")
    return all_findings


# ============================================================================
# REPORTING
# ============================================================================

def print_summary(findings: List[Finding]):
    """Print findings summary."""
    if not findings:
        print("\n✓ No secrets or sensitive variable names detected.")
        return

    # Count by severity
    severity_counts = {"CRITICAL": 0, "HIGH": 0, "MEDIUM": 0, "LOW": 0}
    type_counts = {}
    file_counts = {}

    for f in findings:
        severity_counts[f.severity] = severity_counts.get(f.severity, 0) + 1
        type_counts[f.finding_type] = type_counts.get(f.finding_type, 0) + 1
        file_counts[f.file] = file_counts.get(f.file, 0) + 1

    print("\n" + "=" * 70)
    print("SECRETS SCAN REPORT")
    print("=" * 70)

    print(f"\nTotal Findings: {len(findings)}")
    print(f"  CRITICAL: {severity_counts.get('CRITICAL', 0)}")
    print(f"  HIGH:     {severity_counts.get('HIGH', 0)}")
    print(f"  MEDIUM:   {severity_counts.get('MEDIUM', 0)}")
    print(f"  LOW:      {severity_counts.get('LOW', 0)}")

    print(f"\nBy Type:")
    for ftype, count in sorted(type_counts.items(), key=lambda x: -x[1]):
        print(f"  {ftype}: {count}")

    print(f"\nFiles with findings: {len(file_counts)}")
    for fpath, count in sorted(file_counts.items(), key=lambda x: -x[1])[:15]:
        print(f"  {fpath}: {count} finding(s)")

    print("\n" + "-" * 70)
    print("DETAILED FINDINGS")
    print("-" * 70)

    # Group by severity
    for severity in ["CRITICAL", "HIGH", "MEDIUM", "LOW"]:
        sev_findings = [f for f in findings if f.severity == severity]
        if not sev_findings:
            continue

        print(f"\n[{severity}] — {len(sev_findings)} findings\n")
        for f in sev_findings:
            print(f"  File: {f.file}:{f.line}")
            print(f"  Type: {f.finding_type}")
            print(f"  Match: {f.matched_text}")
            print(f"  Issue: {f.description}")
            print(f"  Fix: {f.remediation}")
            print()


def export_csv(findings: List[Finding], output_path: str):
    """Export findings to CSV file."""
    with open(output_path, "w", newline="", encoding="utf-8") as csvfile:
        writer = csv.writer(csvfile)
        writer.writerow([
            "File", "Line", "Type", "Severity",
            "Matched Text", "Description", "Remediation"
        ])
        for f in findings:
            writer.writerow([
                f.file, f.line, f.finding_type, f.severity,
                f.matched_text, f.description, f.remediation
            ])
    print(f"\nCSV report exported to: {output_path}")


# ============================================================================
# MAIN
# ============================================================================

def main():
    parser = argparse.ArgumentParser(
        description="Scan repository for hardcoded secrets and sensitive variable names (Checkmarx compliance)"
    )
    parser.add_argument(
        "repo_path",
        nargs="?",
        default=".",
        help="Path to the Git repository to scan (defaults to current directory)"
    )

    args = parser.parse_args()
    repo_path = os.path.abspath(args.repo_path)
    repo_name = os.path.basename(repo_path)

    # Run scan
    findings = scan_repository(repo_path)

    # Print summary to console
    print_summary(findings)

    # Always generate CSV report
    output_file = f"secrets_scan_report_{repo_name}.csv"
    export_csv(findings, output_file)

    # Exit code for CI/CD integration
    critical_count = sum(1 for f in findings if f.severity == "CRITICAL")
    if critical_count > 0:
        print(f"\n✗ FAILED: {critical_count} CRITICAL finding(s). Secrets must be remediated.")
        sys.exit(1)
    elif findings:
        print(f"\n⚠ WARNING: {len(findings)} finding(s) detected. Review recommended.")
        sys.exit(0)
    else:
        print("\n✓ PASSED: No findings.")
        sys.exit(0)


if __name__ == "__main__":
    main()
