# search-confluence.ps1
$query = Read-Host "Enter search query"
$baseUrl = "https://cedt-confluence.nam.nsroot.net/confluence"

# Token handling: prompt once, store in env for session
$token = $env:CONFLUENCE_TOKEN
if (-not $token) {
    $token = Read-Host "Enter your Confluence token (will be stored for this session)"
    $env:CONFLUENCE_TOKEN = $token
    Write-Host "Token stored for this session. You won't be prompted again." -ForegroundColor Green
}

$headers = @{ Authorization = "Bearer $token" }

# Use the search/site endpoint which returns results ranked by RELEVANCE
# This is Confluence's native search (same as the UI search box)
$encodedQuery = [uri]::EscapeDataString($query)
$url = "$baseUrl/rest/api/search?cql=siteSearch+~+""$encodedQuery""+AND+type=page&limit=10&expand=content.version,content.body.storage"

Write-Host "Searching for: $query" -ForegroundColor Cyan

try {
    $results = Invoke-RestMethod -Uri $url -Headers $headers
} catch {
    # Fallback: use content search with relevance (no ORDER BY = relevance ranked)
    Write-Host "Trying alternate search..." -ForegroundColor Yellow
    $cql = [uri]::EscapeDataString("type=page AND text ~ ""$query""")
    $url = "$baseUrl/rest/api/content/search?cql=$cql&limit=10&expand=body.storage,version"
    $results = Invoke-RestMethod -Uri $url -Headers $headers
    
    if (-not $results.results -or $results.results.Count -eq 0) {
        Write-Host "No results found." -ForegroundColor Yellow
        exit
    }

    # Sort by relevance (API default) then filter: keep only pages from last 2 years
    $cutoffDate = (Get-Date).AddYears(-2)
    $recentResults = $results.results | Where-Object {
        [datetime]$_.version.when -gt $cutoffDate
    } | Select-Object -First 5

    if (-not $recentResults -or $recentResults.Count -eq 0) {
        Write-Host "No recent results. Showing all results:" -ForegroundColor Yellow
        $recentResults = $results.results | Select-Object -First 5
    }

    Write-Host "Found $($recentResults.Count) results:" -ForegroundColor Green

    $output = @()
    foreach ($page in $recentResults) {
        $bodyText = ($page.body.storage.value -replace '<[^>]+>','')
        $excerptLength = [Math]::Min(1500, $bodyText.Length)
        $links = $page._links

        Write-Host "  - $($page.title) (Modified: $($page.version.when))" -ForegroundColor White

        $output += [PSCustomObject]@{
            Title        = $page.title
            URL          = $baseUrl + $links.webui
            LastModified = $page.version.when
            Excerpt      = $bodyText.Substring(0, $excerptLength)
        }
    }

    $output | ConvertTo-Json | Out-File "confluence-results.json"
    Write-Host "`nDone. Results saved to confluence-results.json" -ForegroundColor Green
    exit
}

# Handle siteSearch response format
$searchResults = $results.results | Select-Object -First 5

Write-Host "Found $($searchResults.Count) results:" -ForegroundColor Green

$output = @()
foreach ($item in $searchResults) {
    $page = $item.content
    $bodyText = ($page.body.storage.value -replace '<[^>]+>','')
    $excerptLength = [Math]::Min(1500, $bodyText.Length)
    $links = $page._links

    Write-Host "  - $($page.title) (Modified: $($page.version.when))" -ForegroundColor White

    $output += [PSCustomObject]@{
        Title        = $page.title
        URL          = $baseUrl + $links.webui
        LastModified = $page.version.when
        Excerpt      = $bodyText.Substring(0, $excerptLength)
    }
}

$output | ConvertTo-Json | Out-File "confluence-results.json"
Write-Host "`nDone. Results saved to confluence-results.json" -ForegroundColor Green
